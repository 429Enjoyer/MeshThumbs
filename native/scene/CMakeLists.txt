cmake_minimum_required(VERSION 3.20)
project(MeshThumbsScene LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MeshThumbs requires a Windows x64 target compiler")
endif()
find_package(Alembic 1.8.8 EXACT CONFIG REQUIRED)
find_package(OpenCASCADE 7.9.3 EXACT CONFIG REQUIRED)
set(OPENNURBS_SOURCE_DIR "" CACHE PATH "Pinned openNURBS source")
set(OPENNURBS_LIBRARY "" CACHE FILEPATH "openNURBS static library")
set(OPENNURBS_ZLIB "" CACHE FILEPATH "openNURBS prefixed zlib static library")
foreach(required OPENNURBS_SOURCE_DIR OPENNURBS_LIBRARY OPENNURBS_ZLIB)
  if(NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
add_library(meshthumbs_scene SHARED alembic.cpp rhino.cpp rhino_tessellation.cpp)
set_target_properties(meshthumbs_scene PROPERTIES PREFIX "")
target_compile_features(meshthumbs_scene PRIVATE cxx_std_17)
target_compile_definitions(meshthumbs_scene PRIVATE ON_CMAKE_BUILD UNICODE _UNICODE)
target_include_directories(meshthumbs_scene PRIVATE "${OPENNURBS_SOURCE_DIR}")
target_link_libraries(meshthumbs_scene PRIVATE Alembic::Alembic "${OPENNURBS_LIBRARY}" "${OPENNURBS_ZLIB}" "${OPENNURBS_LIBRARY}" TKernel TKMath TKBRep TKTopAlgo TKMesh TKGeomBase TKGeomAlgo shlwapi usp10 shell32 ole32 gdi32 rpcrt4 uuid dwrite)
if(MINGW)
  target_compile_options(meshthumbs_scene PRIVATE -UWIN32 -include "${CMAKE_CURRENT_SOURCE_DIR}/compat/opennurbs_mingw.h")
endif()
install(TARGETS meshthumbs_scene RUNTIME DESTINATION .)

option(MESHTHUMBS_SCENE_CHECKS "Build native 3DM geometry regression checks" OFF)
if(MESHTHUMBS_SCENE_CHECKS)
  add_executable(check_rhino check-rhino.cpp rhino_tessellation.cpp)
  target_compile_features(check_rhino PRIVATE cxx_std_17)
  target_compile_definitions(check_rhino PRIVATE ON_CMAKE_BUILD UNICODE _UNICODE)
  target_include_directories(check_rhino PRIVATE "${OPENNURBS_SOURCE_DIR}")
  get_target_property(scene_libraries meshthumbs_scene LINK_LIBRARIES)
  target_link_libraries(check_rhino PRIVATE ${scene_libraries})
  if(MINGW)
    target_compile_options(check_rhino PRIVATE -UWIN32 -include "${CMAKE_CURRENT_SOURCE_DIR}/compat/opennurbs_mingw.h")
  endif()
endif()

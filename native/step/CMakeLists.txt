cmake_minimum_required(VERSION 3.20)
project(MeshThumbsSTEP LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MeshThumbs requires a Windows x64 target compiler")
endif()
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED)
add_library(meshthumbs_step SHARED step.cpp)
target_compile_features(meshthumbs_step PRIVATE cxx_std_17)
target_include_directories(meshthumbs_step PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(meshthumbs_step PRIVATE TKDESTEP TKDEIGES TKMesh TKTopAlgo TKShHealing TKBRep TKMath TKernel)
set_target_properties(meshthumbs_step PROPERTIES PREFIX "")
install(TARGETS meshthumbs_step RUNTIME DESTINATION . LIBRARY DESTINATION .)
foreach(library IN LISTS OpenCASCADE_LIBRARIES)
  # TKExpress is a source-generation tool, not a reader dependency.
  if(NOT library STREQUAL "TKExpress")
    install(FILES "$<TARGET_FILE:${library}>" DESTINATION .)
  endif()
endforeach()
